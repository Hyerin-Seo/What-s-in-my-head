## Proxy 기반 구조적 공유(Structural Sharing) 메모리 참조 맵

```text
🗺️ [Proxy 기반 구조적 공유(Structural Sharing) 메모리 참조 맵]

[Base State (원본 트리)]                       [Next State (새 트리)]
(0x001) Root ──────────────────────────────► (0x999) Root [새 참조 생성 🔴]
  ├─ (0x010) 폴더 1 ────────────────────────► (0x991) 폴더 1 [새 참조 생성 🔴]
  │    ├─ (0x100) 중분류 A ─────────────────► (0x910) 중분류 A [새 참조 생성 🔴]
  │    │    ├─ (0x201) [타겟 노드] ─────────► (0x801) [타겟 노드: "글자 수정"] [새 참조 생성 🔴]
  │    │    └─ (0x202) [형제 노드 나~마] ───► (0x202) [기존 주소 그대로 재사용! 🟢]
  │    └─ (0x101) [중분류 B~Z] ─────────────► (0x101) [기존 주소 그대로 재사용! 🟢]
  └─ (0x020) [폴더 2~10] ───────────────────► (0x020) [기존 주소 그대로 재사용! 🟢]

💡 [구조적 공유의 핵심]
- 수정이 발생한 타겟 노드와 루트까지 거슬러 올라가는 직계 조상 경로만 새 객체 주소(0x9xx)를 발급받음
- 건드리지 않은 99%의 형제·하위 노드들은 원본 메모리 주소(0x1xx, 0x0xx)를 그대로 유지함
➔ React 컴포넌트는 참조가 그대로인 99%의 노드를 렌더링하지 않고 건너뜀 (60fps 유지 비결!)
```

## 순수 스프레드 불변 재귀 방식의 Tab 들여쓰기 코드 (70줄)

```typescript
// ❌ 순수 리액트 스프레드 불변성 방식 : 들여쓰기 하나에 70줄 재귀 탐색 및 수동 슬라이싱
export const indentNodeInTree = (nodes: OutlineNode[], targetId: string): OutlineNode[] => {
  const findAndIndent = (currentNodes: OutlineNode[]): [OutlineNode[], boolean] => {
    const targetIdx = currentNodes.findIndex(n => n.id === targetId);

    if (targetIdx !== -1) {
      if (targetIdx === 0) return [currentNodes, true]; // 첫 불릿 이동 불가

      const targetNode = currentNodes[targetIdx];
      const prevSibling = currentNodes[targetIdx - 1];

      // 이전 형제의 children 배열 끝에 불변 삽입
      const newPrevSibling: OutlineNode = {
        ...prevSibling,
        children: [...(prevSibling.children || []), targetNode],
      };

      // 현재 레벨 배열에서 이전 형제 교체 및 타겟 노드 제거
      const newLevelNodes = [
        ...currentNodes.slice(0, targetIdx - 1),
        newPrevSibling,
        ...currentNodes.slice(targetIdx + 1),
      ];

      return [newLevelNodes, true];
    }

    let modified = false;
    const newNodes = currentNodes.map(node => {
      if (modified || !node.children?.length) return node;
      const [updatedChildren, didModify] = findAndIndent(node.children);
      if (didModify) {
        modified = true;
        return { ...node, children: updatedChildren };
      }
      return node;
    });

    return [newNodes, modified];
  };

  const [result] = findAndIndent(nodes);
  return result;
};
```

splitNode: (targetId, before, after) =>

  transact(() => {
    if (!nodeOf(previousId)) return false;
    const [removed] = apply([{ op: 'delete', id: targetId }]);
    if (!removed.ok) return false;
    apply([{ op: 'replace', id: previousId, text: mergedText }], {
      focusedNodeId: previousId,
      targetCaretOffset: caretOffset,
      selectedNodeIds: [],

  const requests = indentRequests(currentTree(), targetId);

moveNodes: (sourceIds, targetId, position) => {
  const tree = currentTree();
  const topmost = getTopmostIds(tree, sourceIds);
  if (topmost.length === 0) return false;
  if (targetId !== null) {
    if (!getFlatNode(tree, targetId)) return false;

// [patch.ts Draft 엔진] 옛 부모에서 떼고 새 부모에 꽂는 단 두 곳의 원자적 연산
class Draft {
  /** 노드들을 지금 부모의 자식 목록에서 뗀다 (splice/filter) */
  detach(ids: NodeId[]): void {
    for (const [parentId, group] of byParent) {
      const parent = this.get(parentId);
      if (!parent) continue;
      this.set({ ...parent, childIds: parent.childIds.filter((id) => !group.has(id)) });
    }
  }

  /** 새 부모의 지정된 index 자리에 순서대로 단다 (splice 끼워넣기) */
  attach(ids: NodeId[], parentId: NodeId, index: number, touch: boolean): void {
    const parent = this.get(parentId);
    const nextChildIds = [...parent.childIds];
    nextChildIds.splice(safeIndex, 0, ...ids); // 👈 핵심 1줄!
    this.set({ ...parent, childIds: nextChildIds });
  }
}

## 트랜잭션 기반 히스토리 스냅샷 및 타이핑 병합 Undo/Redo 구현 코드

```typescript
// 1) 트랜잭션 : 트리가 실제로 달라졌을 때만 스택에 직전 상태를 Push!
const transact = <T>(fn: () => T): T => {
  const startTree = sync().tree;
  const startView = currentView();
  try {
    return fn();
  } finally {
    if (running.tree !== startTree) {
      // 💡 연속 고속 타이핑은 한 글자마다 쌓지 않고 1초(TYPING_MERGE_MS) 단위로 스마트 병합
      if (!continuesTyping) {
        remember(past, { tree: startTree, view: startView });
        future.length = 0;
      }
      publish(running.tree);
    }
  }
};

// 2) 실행 취소 / 다시 실행 공통 복원 로직
const restore = (from: HistorySnapshot[], to: HistorySnapshot[]): boolean => {
  const snapshot = from.pop();
  if (!snapshot) return false;

  // 반대쪽 스택(undo ➔ redo)으로 현재 상태 이동
  remember(to, { tree: current.tree, view: currentView() });

  // 접힘/펼침 상태는 깨지지 않게 보존하며 트리 복원
  const tree = withFoldStateOf(snapshot.tree, current.tree);
  publish(tree);
  set(snapshot.view);
  return true;
};
```
